import React, { useState } from "react";
import Navbar from "../../components/User/Navbar";
import Footer from "../../components/User/Footer";
import ProfileSideNav from "../../components/User/Profile-Side-Nav";
import { FaSearch, FaStar, FaRegStar, FaTruck, FaChevronLeft, FaChevronRight } from "react-icons/fa";
import { MdRateReview } from "react-icons/md";
import "../../assets/styles/Orders.css";

function Orders() {
  const [searchTerm, setSearchTerm] = useState("");
  const [currentPage, setCurrentPage] = useState(1);
  const ordersPerPage = 3;

  // Sample orders data
  const [orders] = useState([
    {
      id: "002457890KJM",
      status: "Delivered",
      time: "12 Mar, 2026",
      product: "2-Seater Leather Sofa",
      rating: 4.2,
      reviews: 120,
      deliveryDate: "25/04/2020",
      originalPrice: 120,
      discountedPrice: 120,
      quantity: 1,
      image: "https://via.placeholder.com/80x60?text=Sofa",
    },
    {
      id: "002457890KJM",
      status: "Shipped",
      time: "10 Mar, 2026",
      product: "Wooden Coffee Table",
      rating: 4.5,
      reviews: 85,
      deliveryDate: "Expected by 30/04/2020",
      originalPrice: 89,
      discountedPrice: 79,
      quantity: 2,
      image: "https://via.placeholder.com/80x60?text=Table",
    },
    {
      id: "002457890KJM",
      status: "Processing",
      time: "08 Mar, 2026",
      product: "Modern Floor Lamp",
      rating: 4.8,
      reviews: 42,
      deliveryDate: "Expected by 02/05/2020",
      originalPrice: 45,
      discountedPrice: 45,
      quantity: 1,
      image: "https://via.placeholder.com/80x60?text=Lamp",
    },
    {
      id: "002457890KJM",
      status: "Delivered",
      time: "05 Mar, 2026",
      product: "Cotton Bedsheet Set",
      rating: 4.0,
      reviews: 200,
      deliveryDate: "Delivered on 20/03/2020",
      originalPrice: 35,
      discountedPrice: 29,
      quantity: 3,
      image: "https://via.placeholder.com/80x60?text=Bedsheet",
    },
    {
      id: "002457890KJM",
      status: "Cancelled",
      time: "01 Mar, 2026",
      product: "Ceramic Flower Vase",
      rating: 3.9,
      reviews: 15,
      deliveryDate: "Cancelled",
      originalPrice: 25,
      discountedPrice: 25,
      quantity: 1,
      image: "https://via.placeholder.com/80x60?text=Vase",
    },
  ]);

  // Filter orders based on search
  const filteredOrders = orders.filter(order =>
    order.product.toLowerCase().includes(searchTerm.toLowerCase()) ||
    order.id.toLowerCase().includes(searchTerm.toLowerCase())
  );

  // Pagination logic
  const indexOfLastOrder = currentPage * ordersPerPage;
  const indexOfFirstOrder = indexOfLastOrder - ordersPerPage;
  const currentOrders = filteredOrders.slice(indexOfFirstOrder, indexOfLastOrder);
  const totalPages = Math.ceil(filteredOrders.length / ordersPerPage);

  const handlePageChange = (pageNumber) => {
    setCurrentPage(pageNumber);
  };

  // Helper to render star ratings
  const renderStars = (rating) => {
    const fullStars = Math.floor(rating);
    const hasHalfStar = rating % 1 >= 0.5;
    const stars = [];
    
    for (let i = 0; i < fullStars; i++) {
      stars.push(<FaStar key={`star-${i}`} className="text-warning me-1" size={12} />);
    }
    if (hasHalfStar) {
      stars.push(<FaStar key="half-star" className="text-warning me-1" size={12} style={{ opacity: 0.6 }} />);
    }
    const emptyStars = 5 - stars.length;
    for (let i = 0; i < emptyStars; i++) {
      stars.push(<FaRegStar key={`empty-${i}`} className="text-secondary me-1" size={12} />);
    }
    return stars;
  };

  // Get status badge class
  const getStatusBadgeClass = (status) => {
    switch (status.toLowerCase()) {
      case "delivered":
        return "status-delivered";
      case "shipped":
        return "status-shipped";
      case "processing":
        return "status-processing";
      case "cancelled":
        return "status-cancelled";
      default:
        return "status-default";
    }
  };

  return (
    <>
      <Navbar />
      <div className="orders-container container py-3 my-2">
        <h4 className="mb-3 fw-bold">Settings and Profile</h4>
        <div className="row justify-content-center">
          {/* Sidebar */}
          <div className="col-lg-4 col-md-5 mb-3 d-none d-lg-block">
            <ProfileSideNav />
          </div>

          {/* Main Content - My Orders */}
          <div className="col-lg-8 col-md-7">
            <div className="orders-card">
              <div className="orders-header">
                <div>
                  <h4 className="fw-bold mb-0">My Orders</h4>
                  <p className="orders-subtitle">View my Last Purchase History</p>
                </div>
              </div>

              {/* Search Bar */}
              <div className="orders-search-wrapper">
                <div className="orders-search">
                  <FaSearch className="search-icon" />
                  <input
                    type="text"
                    className="search-input"
                    placeholder="Search your products"
                    value={searchTerm}
                    onChange={(e) => {
                      setSearchTerm(e.target.value);
                      setCurrentPage(1);
                    }}
                  />
                </div>
              </div>

              {/* Orders List */}
              <div className="orders-list">
                {currentOrders.length > 0 ? (
                  currentOrders.map((order, index) => (
                    <div key={index} className="order-item">
                      <div className="order-header-row">
                        <div className="order-id-wrapper">
                          <span className="order-id-label">Order ID</span>
                          <span className="order-id-value">{order.id}</span>
                        </div>
                        <div className="order-status-time">
                          <span className={`order-status-badge ${getStatusBadgeClass(order.status)}`}>
                            Status: {order.status}
                          </span>
                          <span className="order-time">Time: {order.time}</span>
                        </div>
                      </div>

                      <div className="order-product-row">
                        <div className="product-image">
                          <img src={order.image} alt={order.product} />
                        </div>
                        <div className="product-details">
                          <h5 className="product-title">{order.product}</h5>
                          <div className="product-rating">
                            {renderStars(order.rating)}
                            <span className="rating-count">({order.reviews})</span>
                          </div>
                          <p className="delivery-date">
                            {order.deliveryDate.includes("Delivered") || order.deliveryDate.includes("Expected") 
                              ? order.deliveryDate 
                              : `Delivered on ${order.deliveryDate}`}
                          </p>
                        </div>
                        <div className="product-price-section">
                          <div className="price-row">
                            <span className="current-price">${order.discountedPrice}</span>
                            {order.originalPrice > order.discountedPrice && (
                              <span className="original-price">${order.originalPrice}</span>
                            )}
                          </div>
                          <p className="quantity">Qty: {order.quantity}</p>
                        </div>
                      </div>

                      <div className="order-actions">
                        <button className="track-order-btn">
                          <FaTruck className="me-2" />
                          Track order
                        </button>
                        <button className="rate-product-btn">
                          <MdRateReview className="me-2" />
                          Rate Your Product
                        </button>
                      </div>
                    </div>
                  ))
                ) : (
                  <div className="no-orders">
                    <p>No orders found matching your search.</p>
                  </div>
                )}
              </div>

              {/* Pagination */}
              {totalPages > 1 && (
                <div className="pagination-wrapper">
                  <button
                    className="pagination-btn"
                    onClick={() => handlePageChange(currentPage - 1)}
                    disabled={currentPage === 1}
                  >
                    <FaChevronLeft />
                  </button>
                  {[...Array(totalPages)].map((_, idx) => (
                    <button
                      key={idx}
                      className={`pagination-number ${currentPage === idx + 1 ? "active" : ""}`}
                      onClick={() => handlePageChange(idx + 1)}
                    >
                      {idx + 1}
                    </button>
                  ))}
                  <button
                    className="pagination-btn"
                    onClick={() => handlePageChange(currentPage + 1)}
                    disabled={currentPage === totalPages}
                  >
                    <FaChevronRight />
                  </button>
                </div>
              )}
            </div>
          </div>
        </div>
      </div>
      <Footer />
    </>
  );
}

export default Orders;
